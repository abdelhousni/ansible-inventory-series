== 4. --list gives variables per host; --export, per group

$ ansible-inventory --list --yaml --limit db1
all:
  children:
    backup:
      hosts:
        db1: {}
    postgresql:
      children:
        db:
          hosts:
            db1:
              ansible_connection: local
              ansible_python_interpreter: '{{ ansible_playbook_python }}'
              postgresql_server_conf:
                max_connections: 200
                port: 5432
              postgresql_version: '15'

$ ansible-inventory --list --yaml --limit db1 --export
all:
  children:
    app:
      vars:
        podman_firewall:
        - port: 8080/tcp
          state: enabled
        podman_registries_conf:
          unqualified-search-registries:
          - registry.example.org
    backup:
      hosts:
        db1: {}
    postgresql:
      children:
        db:
          hosts:
            db1: {}
          vars:
            postgresql_version: '15'
      vars:
        postgresql_server_conf:
          max_connections: 200
          port: 5432
        postgresql_version: '16'
  vars:
    ansible_connection: local
    ansible_python_interpreter: '{{ ansible_playbook_python }}'
